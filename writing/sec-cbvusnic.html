方案：@名单库-方案

Redis中的key设计

  • redis key: risk:t:{obj_type}:{obj_value}
  • field: {tag_id}
  • value: '{expired}|{updated}' (增加字段则按顺序通过竖线分割- 为了降低Redis存储压力)

    其中expired是秒级绝对时间戳，updated是毫秒级绝对时间戳。

示例：

key = risk:t:uid:123
1) "1001"
2) "1730271374|1730271300"

并且为了进一步降低Redis存储压力，在进行Redis写入时，做了一层objType的转换，如下：

private static final ImmutableMap<String, String> TAG_TYPE_CONVERSION_MAP = ImmutableMap.of(
        "userId", "uid", // 将userId 压缩为 uid
        "userIp", "uip" // 将userIp 压缩为 uip
);
// 组装Redis Key
private String assembleRedisTagKey(String objType, String objValue) {
    //进行一次主体 type 转换,降低 redis 存储(只有对 redis 的读写需要转换)
    String storeType = TAG_TYPE_CONVERSION_MAP.get(objType);
    if (StringUtils.isEmpty(storeType)) {
        storeType = objType;
    }
    return String.format("risk:t:%s:%s", storeType, objValue);
}

1.CacheLoader缓存加载 & 数据处理

项目启动时，将标签元数据(TagMeta)从Redis读出到缓存，组装成所需的数据结构,支撑后续功能。

共组装成两个核心数据结构：

  • parentTagMetaMap：Map<tag_id,List<TagMeta>(包含其本身及其所有父标签的列表)>
  • childTagMetaMap： Map<tag_id,List<TagMeta>(包含其本身及其所有子标签的列表-)>

举例：

若此时存在以下五个标签，以<parent_tag_id,tag_id>的形式精简标识，有：①<0,1> ②<1,9> ③<1,87> ④<9,901> ⑤<901,9001>

则在经过处理后，有如下数据结构：

parentTagMetaMap：<1，[①]>、<9，[②,①]>、<87，[③,①]>、<901，[④,②,①]>、<9001，[⑤,④,②,①]>

childTagMetaMap：<1，[①,②,③,④,⑤]>、<9，[②,④,⑤]>、<87，[③]>、<901，[④,⑤]>、<9001,[⑤]>

public class TagMetaCacheImpl implements TagMetaCache, CacheLoader {

    private Map<Integer, TagMetaModel> tagMetaMap; // Map<tag_id,TagMeta>
    private Map<Integer, List<TagMetaModel>> parentTagMetaMap; //  Map<tag_id,List<TagMeta>(包含其本身及其所有父标签的列表)>
    private Map<Integer, List<TagMetaModel>> childTagMetaMap; // Map<tag_id,List<TagMeta>(包含其本身及其所有子标签的列表-)>

    @Autowired
    private TagListRedisService tagListRedisService;
    
    @Override
    public TagMetaModel getTagMetaById(Integer tagId) {
        return tagMetaMap.get(tagId);
    }

    @Override
    public List<TagMetaModel> getParentTagsById(Integer tagId) {
        return parentTagMetaMap.get(tagId);
    }

    @Override
    public List<TagMetaModel> getChildTagsById(Integer tagId) {
        return childTagMetaMap.get(tagId);
    }

    @Override
    public String getCacheName() {
        return "tag_meta";
    }

    @Override
    public int getDelayTime() {
        return 10;
    }

    @Override
    public Boolean reload() {
        try {
            String originValue = tagListRedisService.get(APPMeta.RISK_TAG_META_CACHE_KEY);
            if (StringUtils.isEmpty(originValue)) {
                log.error("event field meta load fail, empty meta");
                tagMetaMap = new HashMap<>();
                return true;
            }
            String value = GzipUtils.unGzip(originValue); // 解压
            if (StringUtils.isEmpty(value)) {
                log.error("event field load fail, gzip error");
                return false;
            }
            List<TagMetaModel> tagMetaModels = Jacksons.BASE.parseList(value, TagMetaModel.class);
            Map<Integer, TagMetaModel> tempTagMetaMap = new HashMap<>(); // Map<tag_id,TagMeta>
            Map<Integer, List<TagMetaModel>> parentTagMap = new HashMap<>(); // Map<parent_tag_id,List<TagMeta>>
            for (TagMetaModel tagMetaModel : tagMetaModels) {
                tempTagMetaMap.put(tagMetaModel.getTagId(), tagMetaModel);
                parentTagMap.computeIfAbsent(tagMetaModel.getParentTagId(), k -> new ArrayList<>()).add(tagMetaModel);
            }
            childTagMetaMap = createChildTagMap(parentTagMap, tempTagMetaMap);
            parentTagMetaMap = createParentTagMap(tempTagMetaMap);
            tagMetaMap = tempTagMetaMap;
            return true;
        } catch (Exception e) {
            log.error("reload tagMeta error: {}", e.toString());
            return false;
        }
    }

    private Map<Integer, List<TagMetaModel>> createParentTagMap(Map<Integer, TagMetaModel> allTags) {
        Map<Integer, List<TagMetaModel>> tempParentTagMap = new HashMap<>();
        for (TagMetaModel tag : allTags.values()) {
            List<TagMetaModel> result = new ArrayList<>();
            int currentTagId = tag.getTagId();
            while (true) {
                TagMetaModel currentTag = allTags.get(currentTagId);
                if (currentTag == null) {
                    break;
                }
                result.add(currentTag);
                if (currentTag.getParentTagId() == 0) { // 根节点就终止
                    break;
                }
                currentTagId = currentTag.getParentTagId(); // 向上找父标签
            }
            tempParentTagMap.put(tag.getTagId(), result);
        }
        return tempParentTagMap; // Map<tag_id,List<TagMeta>(包含其本身及其所有父标签的列表)>
    }

    private Map<Integer, List<TagMetaModel>> createChildTagMap(Map<Integer, List<TagMetaModel>> parentTagMap, Map<Integer, TagMetaModel> tempTagMetaMap) {
        Map<Integer, List<TagMetaModel>> tempChildTagMap = new HashMap<>();
        for (TagMetaModel tagMetaModel : tempTagMetaMap.values()) {
            List<TagMetaModel> childTagList = new ArrayList<>();
            childTagList.add(tagMetaModel); // 先添加其自身
            List<Integer> tagIds = new ArrayList<>();
            tagIds.add(tagMetaModel.getTagId());
            int deep = 0;
            while (deep < tagIds.size()) {
                Integer tagId = tagIds.get(deep);
                List<TagMetaModel> childTags = parentTagMap.get(tagId); // 添加其名下所有子标签
                if (childTags != null) {
                    for (TagMetaModel childTag : childTags) {
                        tagIds.add(childTag.getTagId());
                        childTagList.add(childTag);
                    }
                }
                deep++;
            }
            tempChildTagMap.put(tagMetaModel.getTagId(), childTagList);
        }
        return tempChildTagMap; // Map<tag_id,List<TagMeta>(包含其本身及其所有子标签的列表)>
    }
}

2.核心接口分析

若当前有以下的标签层级关系：

1.queryTags -返回命中的未过期标签或其子标签

📌

此接口会将过期的标签过滤掉，只返回未过期(生效)的标签。

此接口一般在“业务侧”使用，业务方一般仅关注是否命中传入的标签。

  • 接口含义：判定名单objType:objValue是否命中标签tags或其子标签；若是，返回命中的标签。
  • 首先从Redis查出当前risk:t:objType:objValue的所有标签作为taskMap
  • 再补全传入的tags的所有子标签作为childTags
  • 最后取taskMap与childTags的交集，作为返回结果
  • 返回空列表 -> 未命中tags中的任何标签。
  • 返回非空列表 -> 返回的是命中的tags标签或其子标签。
  • 例1：若此时的tags={6,7} Redis中此名单的标签tagMap={2,3}。则根据tags补全后的childTags={6,7},将其与tags求交集后，返回结果为{}(空列表)。
  • 例2：若此时的tags={2,3} Redis中此名单的标签tagMap={6,7}。则根据tags补全后的childTags={2,3,4,5,6,7},将其与tags求交集后，返回结果为{6,7}。
private static final String HASH_VALUE_SPLIT_CHAR = "|";
public ListItem queryTags(String objType, String objValue, List<Integer> tags, long expired) {
    String key = assembleRedisTagKey(objType, objValue); // 组装Redis key
    Map<String, String> tagMap = tagListRedisService.hgetall(key); // 1.从Redis读出此名单的标签Map
    ListItem items = new ListItem();
    items.setObjType(objType);
    items.setObjValue(objValue);
    items.setTags(new ArrayList<>());
    if (tagMap.isEmpty()) {
        return items;
    }
    for (Integer tagId : tags) {
        List<TagMetaModel> childTags = tagMetaCache.getChildTagsById(tagId); // 2.取出每一个标签的所有子标签及其本身
        if (null == childTags) {
            continue;
        }
        for (TagMetaModel childTag : childTags) {
            String tagInfo = tagMap.get(childTag.getTagId().toString());
            if (StringUtils.isNotEmpty(tagInfo)) {
                String[] values = StringUtils.split(tagInfo, HASH_VALUE_SPLIT_CHAR); // 3.“|”分隔，得到expired|updated_at
                if (values.length < 2) {
                    continue;
                }
                long tagExpired = Long.parseLong(values[0]);
                if (tagExpired < expired) { // 4.过滤掉不生效标签（已过期）
                    continue;
                }
                TagItem tagItem = new TagItem();
                tagItem.setTagId(childTag.getTagId());
                tagItem.setExpired(tagExpired);
                tagItem.setUpdated(Long.parseLong(values[1]));
                items.addToTags(tagItem); // 收集结果
            }
        }
    }
    return items;
}

2.queryTagsWithParent -返回命中标签及命中标签的父标签

💡

此接口一般在“服务侧”使用，如：实时风控的规则引擎中，需要该标签所关联的全量父子标签。

且父标签的过期时间由所有子标签决定，从逻辑上来讲，父标签的过期时间 = MAX{son1.expired,son2.expired......}

  • 即：先queryTags，再”由子索父“。

  • 返回空列表 -> 未命中。
  • 返回非空列表 -> 非空列表中的标签就是 "命中的标签+命中标签的父标签"。
  • 若此时的tags={2,3} Redis中此名单的标签tagMap={6,7}。首先此时queryTags之后的tagItems={6,7}，再递归向上寻找父标签，最终结果为tagItems={1,3,6,7}
@Override
public ListItem queryTagsWithParent(String objType, String objValue, List<Integer> tags, long expired) {
    ListItem tagItems = queryTags(objType, objValue, tags, expired); // 1.查出此名单当前生效的所有子标签及其本身
    Map<Integer, TagItem> tagWithParentMap = new HashMap<>();
    for (TagItem tagItem : tagItems.getTags()) {
        List<TagMetaModel> parentTags = tagMetaCache.getParentTagsById(tagItem.getTagId()); // 2.为每个标签项查找其父标签（不考虑标签过期）
        if (null == parentTags) {
            continue;
        }
        for (TagMetaModel parentTag : parentTags) {
            if (tagWithParentMap.containsKey(parentTag.getTagId())) {
                continue;
            }
            tagWithParentMap.put(parentTag.getTagId(), new TagItem().setTagId(parentTag.getTagId())); // 3.
        }
    }
    tagItems.setTags(new ArrayList<>(tagWithParentMap.values()));
    return tagItems;
}

3.addTags - 新增名单

  1. 写入Redis，以hash的数据结构存储。
  1. Holo：名单库数据的持久化，实现”追加写“操作，并在同步时做了一步特殊处理：‘source="风控平台"的数据不需要更新到 holo，风控平台 后台写标签会主动写holo’
  2. Iceberg：持久化日志，实时同步操作日志，将数据分区存储。
@Override
public boolean addTags(
        String objType, String objValue, List<TagItem> tags, String reason, String operator, String source
) {
    String tagKey = assembleRedisTagKey(objType, objValue);
    Map<String, String> tagMap = new HashMap<>();
    for (TagItem tag : tags) { // 1.组装为Map
        tagMap.put(String.valueOf(tag.getTagId()), tag.getExpired() + HASH_VALUE_SPLIT_CHAR + tag.getUpdated());
    }
    boolean success = tagListRedisService.hmset(tagKey, tagMap); // 2.存入Redis
    if (success) {
        // 3.写入Talos-优先保障写 redis 成功
        sendTagListDatas(objType, objValue, tags, reason, operator, source);
    }
    return success;
}
