方案文档：@标签库-方案

20241127和峰哥讨论了一下，记录一下几个点：

  1. 开发顺序：标签库(Holo)->标签库(Mysql)

  1. 原因：核心服务读取标签库信息时，是从Redis中读取的，保证Redis的数据写入成功。

建表

CREATE TABLE `tag_meta` (
  `id` int(11) unsigned NOT NULL AUTO_INCREMENT COMMENT 'id',
  `tag_id` int(11) NOT NULL DEFAULT '0' COMMENT '标签 ID',
  `tag_name` varchar(64) NOT NULL DEFAULT '' COMMENT '标签名称',
  `parent_tag_id` int(11) NOT NULL DEFAULT '0' COMMENT '父标签 id',
  `status` tinyint(3) NOT NULL DEFAULT '1' COMMENT '标签状态 1: online 0: offline',
  `description` varchar(256) NOT NULL DEFAULT '' COMMENT '标签描述',
  `updated_by` varchar(64) NOT NULL DEFAULT 'risk' COMMENT '更新人',
  `updated_at` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_tid_ptid` (`tag_id`,`parent_tag_id`),
  KEY `idx_pid` (`parent_tag_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='标签元信息'

BEGIN;
CREATE TABLE hologres_tjwqstaging_hdd.risk_risk.tag_list (
    obj_type text,
    obj_value text,
    tag_id int4,
    expired int8,
    reason text,
    updated_by text,
    updated_at TIMESTAMP DEFAULT now(),
    PRIMARY KEY (obj_type, obj_value, tag_id)
);
call set_table_property('risk_risk.tag_list', 'orientation', 'row');
COMMIT;
