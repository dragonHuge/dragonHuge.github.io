Spark 离线任务规则

2026-07-09"本地沉淀 · 风控工程知识库"

本文是工程规则（团队约定 + 踩坑结论），不是入门教程： 适合「已经知道要做什么，只想知道怎么做才不出事」的场景。

1. 适用范围

数据工场 Spark Jar 作业（非 Spark SQL / Notebook）。典型场景：离线扫源表 + 调下游 RPC / 直写 Redis 做兜底清理 / 回灌。

平台运行时：Spark 3.1.x + JDK 8。

2. 版本 / 依赖

坐标scope备注
org.apache.spark:spark-core_${scala.binary.version}provided平台提供
org.apache.spark:spark-sql_${scala.binary.version}provided平台提供
org.apache.spark:spark-catalyst_${scala.binary.version}provided平台提供
业务 thrift 桩（Java 8 编译版）compile不能用 Java 21 编译的版本
com.公司:apus-client / apus-registry-zookeepercompile调下游 RPC
redis.clients:jediscompile直写 Redis 场景

铁律：

  • maven.compiler.release 必须 8（class 文件版本 52），否则 Spark 3.1 加载报 UnsupportedClassVersionError
  • Spark / Hadoop / Iceberg / Hive 全部 provided，不打进 fat-jar

3. 标准骨架

3.1 Main + args 解析

public final class ${sparkJobClass} {

    private static final Logger LOG = LoggerFactory.getLogger(${sparkJobClass}.class);
    private static final long RPC_TIMEOUT_MS = 3000L;
    private static final String METRIC_PREFIX = "${metricPrefix}.spark_${jobShortName}.";

    private ${sparkJobClass}() {}

    public static void main(String[] args) throws Exception {
        Map<String, String> a = parseArgs(args);

        // 参数（灰度护栏默认安全）
        String snapshotDate = require(a, "snapshotDate");                    // 必填
        String sourceTable = a.getOrDefault("sourceTable", "${defaultSourceTable}");
        boolean dryRun = Boolean.parseBoolean(a.getOrDefault("dryRun", "true"));
        int maxQps = Integer.parseInt(a.getOrDefault("maxQps", "50"));
        long maxTotal = Long.parseLong(a.getOrDefault("maxTotal", "1000"));

        long jobStart = System.currentTimeMillis();
        PerfCounter.setMeterCount(METRIC_PREFIX + "job.start", 1);
        LOG.info("[${sparkJobClass}] snapshotDate={}, sourceTable={}, dryRun={}, maxQps={}, maxTotal={}",
            snapshotDate, sourceTable, dryRun, maxQps, maxTotal);

        try {
            SparkSession spark = SparkSession.builder()
                .appName("${appNamePrefix}-" + snapshotDate)
                .enableHiveSupport()
                .getOrCreate();

            // §3.2 扫源 + 统计
            Dataset<Row> targets = spark.sql(...).repartition(${repartitionNum});
            long targetCount = targets.count();
            LOG.info("[${sparkJobClass}] target rows = {}", targetCount);
            PerfCounter.setMeterCount(METRIC_PREFIX + "rows.target", targetCount);

            // §3.3 分布观测
            // ...

            // §3.4 dryRun 短路
            if (dryRun) {
                PerfCounter.setMeterCount(METRIC_PREFIX + "dry_run", 1);
                LOG.info("[${sparkJobClass}] DRY_RUN: 仅统计，样例(<=20):");
                targets.limit(20).collectAsList().forEach(r -> LOG.info("    {}", r));
                PerfCounter.setMeterCount(METRIC_PREFIX + "job.completed", 1);
                return;
            }

            // §3.5 实跑：限速 + 上限 + foreachPartition
            int numPartitions = targets.rdd().getNumPartitions();
            int perPartitionQps = Math.max(1, maxQps / Math.max(1, numPartitions));
            long perPartitionQuota = (maxTotal + numPartitions - 1) / numPartitions;

            LongAccumulator calledAcc = spark.sparkContext().longAccumulator("called");
            LongAccumulator succeedAcc = spark.sparkContext().longAccumulator("succeed");
            LongAccumulator skipAcc = spark.sparkContext().longAccumulator("skip");
            LongAccumulator failAcc = spark.sparkContext().longAccumulator("fail");

            targets.foreachPartition(new ${partitionFnClass}(
                perPartitionQps, perPartitionQuota,
                calledAcc, succeedAcc, skipAcc, failAcc));

            LOG.info("[${sparkJobClass}] done: target={}, called={}, succeed={}, skip={}, fail={}",
                targetCount, calledAcc.value(), succeedAcc.value(), skipAcc.value(), failAcc.value());
            PerfCounter.setMeterCount(METRIC_PREFIX + "rpc.called", calledAcc.value());
            PerfCounter.setMeterCount(METRIC_PREFIX + "rpc.succeed", succeedAcc.value());
            PerfCounter.setMeterCount(METRIC_PREFIX + "job.completed", 1);
        } catch (Exception e) {
            PerfCounter.setMeterCount(METRIC_PREFIX + "job.failed", 1);
            throw e;
        } finally {
            PerfCounter.setTimerValue(METRIC_PREFIX + "job.all", System.currentTimeMillis() - jobStart);
            // 平台调度的 SparkSession 由平台管理，不主动 stop
        }
    }
}

3.2 Args 解析（key=value）

private static Map<String, String> parseArgs(String[] args) {
    HashMap<String, String> m = new HashMap<>();
    for (String arg : args) {
        int i = arg.indexOf('=');
        if (i > 0) {
            m.put(arg.substring(0, i).trim(), arg.substring(i + 1).trim());
        }
    }
    return m;
}

private static String require(Map<String, String> a, String key) {
    String v = a.get(key);
    if (v == null || v.isEmpty()) {
        throw new IllegalArgumentException("missing required arg: " + key + "=<value>");
    }
    return v;
}

3.3 Partition Function（限速 + 配额）

static final class ${partitionFnClass} implements ForeachPartitionFunction<Row> {
    private final int perPartitionQps;
    private final long perPartitionQuota;
    private final LongAccumulator calledAcc, succeedAcc, skipAcc, failAcc;

    @Override
    public void call(Iterator<Row> rows) throws Exception {
        // executor 端 new RPC client（不可序列化字段）
        ${rpcClientClass} client = new ${rpcClientClass}(RPC_TIMEOUT_MS);
        long minIntervalMs = 1000L / Math.max(1, perPartitionQps);
        long quotaLeft = perPartitionQuota;
        long lastCallMs = 0L;

        while (rows.hasNext()) {
            Row r = rows.next();
            if (quotaLeft <= 0) {
                PerfCounter.setMeterCount(METRIC_PREFIX + "quota_exhausted", 1);
                break;
            }
            // 校验字段非空
            String biz = r.getAs("${businessKey}");
            if (biz == null) {
                PerfCounter.setMeterCount(METRIC_PREFIX + "row.skip_invalid", 1);
                continue;
            }
            // 令牌桶限速
            long now = System.currentTimeMillis();
            long wait = minIntervalMs - (now - lastCallMs);
            if (wait > 0) Thread.sleep(wait);
            lastCallMs = System.currentTimeMillis();

            quotaLeft--;
            calledAcc.add(1);
            try {
                boolean ok = client.${rpcMethod}(biz, /* ... */);
                if (ok) succeedAcc.add(1); else skipAcc.add(1);
            } catch (Exception e) {
                failAcc.add(1);
                LOG.warn("[${sparkJobClass}] call failed, biz={}", biz, e);
            }
        }
    }
}

3.4 Shade 打包（隔离 guava / thrift 冲突）

spark/pom.xml：

<plugin>
    <groupId>org.apache.maven.plugins</groupId>
    <artifactId>maven-shade-plugin</artifactId>
    <version>3.5.1</version>
    <executions>
        <execution>
            <phase>package</phase>
            <goals><goal>shade</goal></goals>
            <configuration>
                <createDependencyReducedPom>false</createDependencyReducedPom>
                <filters>
                    <filter>
                        <artifact>*:*</artifact>
                        <excludes>
                            <exclude>META-INF/*.SF</exclude>
                            <exclude>META-INF/*.DSA</exclude>
                            <exclude>META-INF/*.RSA</exclude>
                        </excludes>
                    </filter>
                </filters>
                <artifactSet>
                    <excludes>
                        <exclude>org.apache.spark:*</exclude>
                        <exclude>org.scala-lang:*</exclude>
                        <exclude>org.apache.hadoop:*</exclude>
                        <exclude>org.apache.iceberg:*</exclude>
                        <exclude>org.apache.hive:*</exclude>
                    </excludes>
                </artifactSet>
                <relocations>
                    <!-- guava 隔离：Spark/Hadoop 自带旧 guava，本作业的新 guava 会冲突 -->
                    <relocation>
                        <pattern>com.google.common</pattern>
                        <shadedPattern>${basePackage}.shaded.guava</shadedPattern>
                    </relocation>
                    <!-- thrift 隔离：平台 spark 自带 thrift 可能冲突 -->
                    <relocation>
                        <pattern>org.apache.thrift</pattern>
                        <shadedPattern>org.shaded.thrift</shadedPattern>
                        <excludes>
                            <exclude>org.codehaus.plexus.util.xml.Xpp3Dom</exclude>
                            <exclude>org.codehaus.plexus.util.xml.pull.*</exclude>
                        </excludes>
                    </relocation>
                </relocations>
            </configuration>
        </execution>
    </executions>
</plugin>

4. 配置项

配置 key类型默认说明
snapshotDatearg（必填）—T-1 分区，INT yyyyMMdd
sourceTablearg由使用方提供源表全限定名（按 region 传）
dryRunargtrue默认安全，只统计
maxQpsarg50全局 RPC 限速
maxTotalarg1000单次运行总上限
nowEpocharg启动时 sysTimeSec作业时刻（影响过期判定）
expireSecondsarg180 * 86400过期阈值，仅清理场景用

铁律：

  • dryRun 默认 true，使用方显式 dryRun=false 才实跑
  • 限速与上限配对使用，单纯限速没上限的话长期任务能跑爆下游

6. 常见踩坑

6.1 🔴 用 Java 21 编译

后果：UnsupportedClassVersionError，作业起不来。 正确：maven.compiler.release=8，本 module 独立维持 JDK8 字节码。

6.2 🔴 RPC client 当 broadcast variable

错误：把 thrift client 当字段持有后 broadcast 到 executor。 后果：client 不可序列化，作业失败。 正确：在 foreachPartition 内部 new client（§3.3）。

6.3 🔴 dryRun 默认 false

后果：误删 / 误写线上数据。 正确：默认 true，使用方显式改 false 才实跑。

6.4 🟠 限速没上限

现象：业务高峰期 maxQps 没生效，作业整体跑爆下游。 正确：限速 + 配额（maxTotal）配对使用。

6.5 🟠 fat-jar 打进 spark / hadoop

后果：jar 几百 MB，提交慢；可能与平台版本冲突。 正确：provided + shade artifactSet excludes。

6.6 🟠 SparkSession 主动 stop

错误：作业末尾 spark.stop()。 后果：平台调度后续清理失败。 正确：由平台管理，不主动 stop。

6.7 🟢 没分区分布观测

对清理 / 回灌类作业，必须按业务维度 groupBy 统计（如 level / type），便于 dryRun 时判断是否符合预期。
