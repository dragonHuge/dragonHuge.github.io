多Agent项目：智能问诊AI客服系统

项目简介

智能问诊客服系统是一个基于 LangChain 1.0.3 + LangGraph 1.0.2 构建的多智能体问诊助手系统，通过自然语言交互实现导诊、咨询、诊断建议等功能。

核心功能

- ✅ 多智能体协作：私人问诊顾问（supervisor）、体检员、药师四个智能体协同工作

- ✅ 长短期记忆：基于 PostgreSQL 的双层记忆系统

- ✅ 文档管理：PDF上传、智能提取信息

- ✅ 用户认证：JWT Token 认证

- ✅ 会话管理：多会话支持、历史记录查询

- ✅ 联网搜索：集成 Tavily 搜索工具

- ✅ LangSmith 追踪：完整的可观测性支持

1. 状态定义 ([agents/state.py]

from typing import TypedDict, List, Optional, Annotated
from langgraph.graph.message import add_messages


class MedicalAgentState(TypedDict):
    """医疗智能体系统的共享状态"""
    
    # 消息列表（自动累积！核心设计）
    messages: Annotated[list, add_messages]
    
    # 用户 ID
    user_id: str
    
    # 会话线程 ID
    thread_id: str
    
    # 医疗文档列表
    medical_documents: List
    
    # 诊断结果（来自主治医生）
    diagnosis: Optional[str]
    
    # 处方信息（来自药师）
    prescription: Optional[str]
    
    # 下一步要执行的节点
    next: Optional[str]

核心设计：

  • Annotated[list, add_messages]：这是 LangGraph 的核心机制，表示多个节点返回消息时，会自动追加而不是覆盖历史

2. 主工作流图构建 ([agents/multi_agent.py]

from langgraph.graph import StateGraph, END
from app.agents.state import MedicalAgentState
from app.agents.memory import (
    init_memory_system,
    get_short_term_memory,
    get_long_term_memory
)
from app.agents.supervisor import supervisor_node
from app.agents.medical_examiner import medical_examiner_node
from app.agents.attending_doctor import attending_doctor_node
from app.agents.pharmacist import pharmacist_node


def create_medical_agent_system():
    """创建医疗智能体系统（中心辐射式架构）"""
    
    # 初始化记忆系统
    init_memory_system()
    checkpointer = get_short_term_memory()  # PostgresSaver
    store = get_long_term_memory()          # PostgresStore
    
    # 构建图
    builder = StateGraph(MedicalAgentState)
    
    # 添加所有节点
    builder.add_node("supervisor", supervisor_node)
    builder.add_node("attending_doctor", attending_doctor_node)
    builder.add_node("medical_examiner", medical_examiner_node)
    builder.add_node("pharmacist", pharmacist_node)
    
    # 设置入口
    builder.set_entry_point("supervisor")
    
    # 添加条件路由（Supervisor 决策下一步）
    builder.add_conditional_edges(
        "supervisor",
        lambda state: state.get("next", "__end__"),
        {
            "attending_doctor": "attending_doctor",
            "medical_examiner": "medical_examiner",
            "pharmacist": "pharmacist",
            "__end__": END
        }
    )
    
    # 所有子节点处理完直接结束（一问一答模式）
    builder.add_edge("attending_doctor", END)
    builder.add_edge("medical_examiner", END)
    builder.add_edge("pharmacist", END)
    
    # 编译图（传入检查点和存储）
    graph = builder.compile(checkpointer=checkpointer, store=store)
    
    return graph

def supervisor_node(state: MedicalAgentState):
    """私人医疗顾问 = 监督者 + 导诊"""
    
    # 1. 获取 LLM 和记忆
    model = get_model()
    messages = state["messages"]
    user_id = state["user_id"]
    store = get_long_term_memory()
    
    # 2. 获取用户长期记忆（个人信息、医疗历史）
    user_context = get_user_long_memory(user_id, store)
    
    # 3. 构建路由提示词
    memory_section = ""
    if user_context:
        memory_section = f"\n=== 用户历史信息 ===\n{user_context}\n=== 历史信息结束 ==="
    
    route_prompt = ROUTE_PROMPT.format(memory_section=memory_section)
    route_messages = [SystemMessage(content=route_prompt)] + messages
    
    # 4. LLM 路由决策
    route_response = model.invoke(route_messages)
    route_text = route_response.content.strip().lower()
    
    # 5. 解析路由结果
    valid_routes = ["medical_examiner", "attending_doctor", "pharmacist", "supervisor"]
    next_node = "supervisor"
    for r in valid_routes:
        if r in route_text:
            next_node = r
            break
    
    print(f"[路由决策] LLM决策结果: {route_text} -> {next_node}")
    
    # 6. 路由到其他智能体，或者自己处理
    if next_node != "supervisor":
        return {"next": next_node}
    
    # 7. 自己处理（问候、闲聊、导诊）
    agent = create_agent(
        model=model,
        tools=[search_web],
        system_prompt=SUPERVISOR_PROMPT.format(memory_section=memory_section)
    )
    
    config = {"configurable": {"user_id": user_id, "thread_id": state["thread_id"]}}
    response = agent.invoke({"messages": messages}, config=config)
    
    # 8. 添加角色前缀
    if response["messages"] and isinstance(response["messages"][-1], AIMessage):
        last_msg = response["messages"][-1]
        if not last_msg.content.startswith("【私人医疗顾问】"):
            last_msg.content = "【私人医疗顾问】\n" + last_msg.content
    
    return {
        "messages": response["messages"],
        "next": "__end__"
    }

def get_user_long_memory(user_id: str, store=None) -> str:
    """获取用户的长期记忆（个人信息、医疗历史等）"""
    if store is None:
        store = get_long_term_memory()
    
    user_info = []
    
    # 获取用户基本信息（按字段去重）
    preferences = store.search(("user_preferences", user_id))
    if preferences:
        pref_dict = {}
        for item in preferences:
            key_parts = item.key.split('|') if '|' in item.key else [item.key]
            item_id = key_parts[-1]
            pref_dict[item_id] = item.value
        
        pref_text = "\n".join([
            f"{value.get('key')}: {value.get('value')}"
            for value in pref_dict.values()
        ])
        user_info.append("【用户基本信息】\n" + pref_text)
    
    # 获取用户医疗历史（按类别分组）
    medical_history = store.search(("user_medical_history", user_id))
    if medical_history:
        medical_by_category = {}
        for item in medical_history:
            category = item.value.get("category", "unknown")
            content = item.value.get("content", "")
            if category not in medical_by_category:
                medical_by_category[category] = []
            medical_by_category[category].append(content)
        
        history_lines = []
        for category, contents in medical_by_category.items():
            if len(contents) == 1:
                history_lines.append(f"{category}: {contents[0]}")
            else:
                items_text = "\n".join([f"  {i+1}. {c}" for i, c in enumerate(contents)])
                history_lines.append(f"{category}:\n{items_text}")
        
        medical_text = "\n".join(history_lines)
        user_info.append("【医疗历史】\n" + medical_text)
    
    return "\n\n".join(user_info) if user_info else ""

记忆命名空间设计：

  • ("user_preferences", user_id)：用户基本信息（姓名、年龄、性别等）
  • ("user_medical_history", user_id)：医疗历史（过敏史、手术史、症状等）
  • ("user_medical_records", user_id, type)：医疗文档（诊断、处方等）

    if conversation:

        db.commit()

    }

            "thread_id": request.thread_id
        }
